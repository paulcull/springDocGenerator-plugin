    Code Examples¶

    This guide provides practical examples of using our API with different programming languages and use cases.

    Basic Examples¶

    Authentication¶

    # Python
from api_client import ApiClient

client = ApiClient()
client.configure_auth("YOUR_API_KEY")

    // Java
import com.example.client.ApiClient;

ApiClient client = new ApiClient();
client.setApiKey("YOUR_API_KEY");

    // TypeScript
import { ApiClient } from '@example/api-client';

const client = new ApiClient({
  apiKey: 'YOUR_API_KEY'
});

    Making API Calls¶

    # Python
response = client.resources.list()
for item in response.data:
    print(f"ID: {item.id}, Name: {item.name}")

    // Java
ResourceResponse response = client.resources().list();
response.getData().forEach(item -> {
    System.out.printf("ID: %s, Name: %s%n", item.getId(), item.getName());
});

    // TypeScript
const response = await client.resources.list();
response.data.forEach(item => {
  console.log(`ID: ${item.id}, Name: ${item.name}`);
});

    Pagination¶

    # Python
page = 1
while True:
    response = client.resources.list(page=page, limit=100)
    if not response.data:
        break
    for item in response.data:
        process_item(item)
    page += 1

    Error Handling¶

    # Python
try:
    response = client.resources.get("invalid-id")
except ApiError as e:
    if e.status_code == 404:
        print("Resource not found")
    else:
        print(f"API error: {e.message}")

    Rate Limiting¶

    # Python
from time import sleep
from api_client.exceptions import RateLimitError

def make_request_with_retry(func, max_retries=3):
    for attempt in range(max_retries):
        try:
            return func()
        except RateLimitError as e:
            if attempt == max_retries - 1:
                raise
            sleep(e.retry_after)

    Batch Operations¶

    # Python
def batch_process(items, batch_size=100):
    for i in range(0, len(items), batch_size):
        batch = items[i:i + batch_size]
        client.resources.batch_create(batch)

    Testing Examples¶

    Unit Tests¶

    # Python
import unittest
from unittest.mock import Mock

class TestApiClient(unittest.TestCase):
    def setUp(self):
        self.client = ApiClient()
        self.client.resources = Mock()

    def test_list_resources(self):
        self.client.resources.list.return_value = {
            'data': [{'id': '1', 'name': 'Test'}]
        }
        response = self.client.resources.list()
        self.assertEqual(len(response['data']), 1)

    Integration Tests¶

    # Python
import os
import unittest

class TestApiIntegration(unittest.TestCase):
    def setUp(self):
        self.client = ApiClient()
        self.client.configure_auth(os.getenv('API_KEY'))

    def test_create_and_delete_resource(self):
        # Create resource
        resource = self.client.resources.create({
            'name': 'Test Resource'
        })
        self.assertIsNotNone(resource['id'])

        # Delete resource
        self.client.resources.delete(resource['id'])

    Best Practices¶

    1. Always use environment variables for API keys
    2. Implement proper error handling
    3. Use pagination for large datasets
    4. Handle rate limiting with retries
    5. Log API responses for debugging
    6. Use appropriate timeout values
    7. Validate input data before sending
